const [isPriorityOpen, setIsPriorityOpen] = useState(false);

**Reset function:**

  setDueDate("");
  setIsExpanded(false);
  setIsPriorityOpen(false);

**Submit function:**
```typescript
const handleSubmit = useCallback(() => {
  if (name.trim().length === 0) return; // Pitfall 7: no empty name submission
  onCreate({
    name: name.trim(),
    priority,
    due_date: dueDate || null,
  });
  resetForm();
}, [name, priority, dueDate, onCreate, resetForm]);
```

**Collapsed state (default, per D-08, UI-SPEC):**
Use `hover:bg-accent` (NOT `hover:bg-neutral-800` -- FOUN-02 prohibits raw palette classes in component files):
```tsx
<div
  className="flex items-center gap-2 px-4 min-h-[44px] cursor-pointer hover:bg-accent transition-colors"
  onClick={() => {
    setIsExpanded(true);
    // autoFocus on name input handles focus after render
  }}
  onKeyDown={(e) => {
    // Type-to-expand: if printable character, expand and seed name (Pattern 6)
    if (e.key.length === 1 && !e.ctrlKey && !e.metaKey && !e.altKey) {
      setName(e.key);
      setIsExpanded(true);
    }
  }}
  tabIndex={0}
  role="button"
  aria-label="Add a new todo"
>
  <Plus className="text-muted-foreground" />
  <span className="text-sm italic text-muted-foreground">Add a todo...</span>
</div>
```

Note: The Plus icon is standalone (not inside a Button or other shadcn component), so the icons.md sizing rule does not strictly apply. However, for consistency, omit sizing classes and let the icon render at its default size.

**Expanded state (active entry, per D-09 through D-12, UI-SPEC):**
Use `hover:bg-accent` in popover options (NOT `hover:bg-neutral-800` -- FOUN-02):
```tsx
<div className="flex items-center gap-2 px-4 min-h-[44px]">
  {/* Status icon placeholder -- always not_started for new todos */}
  <div className="size-8 shrink-0" /> {/* spacer matching icon button width */}

  {/* Name input (flex-1, per UI-SPEC) */}
  <Input
    ref={nameInputRef}
    value={name}
    onChange={(e) => setName(e.target.value)}
    onKeyDown={(e) => {
      if (e.key === "Enter") {
        e.preventDefault();
        handleSubmit();
      } else if (e.key === "Escape") {
        e.preventDefault();
        resetForm();
      } else if (e.key === "Tab" && !e.shiftKey) {
        e.preventDefault();
        priorityButtonRef.current?.focus();
      }
    }}
    placeholder="Todo name..."
    autoFocus
    className="h-auto border-0 shadow-none p-0 focus-visible:ring-1 focus-visible:ring-ring/50 text-base flex-1"
    aria-label="New todo name"
  />

  {/* Priority selector (per D-09, UI-SPEC) */}
  <Popover open={isPriorityOpen} onOpenChange={setIsPriorityOpen}>
    <PopoverTrigger
      render={
        <button
          ref={priorityButtonRef}
          className={cn(
            "text-sm px-2 py-1 rounded-sm font-medium shrink-0 cursor-pointer",
            PRIORITY_STYLES[priority]
          )}
          onKeyDown={(e) => {
            if (e.key === "Tab" && !e.shiftKey) {
              e.preventDefault();
              dateInputRef.current?.focus();
            } else if (e.key === "Tab" && e.shiftKey) {
              e.preventDefault();
              nameInputRef.current?.focus();
            } else if (e.key === "Escape") {
              e.preventDefault();
              if (isPriorityOpen) {
                setIsPriorityOpen(false);
              } else {
                resetForm();
              }
            } else if (e.key === "Enter" && !isPriorityOpen) {
              e.preventDefault();
              handleSubmit();
            }
          }}
          aria-haspopup="listbox"
          aria-expanded={isPriorityOpen}
          aria-label="Select priority"
        />
      }
    >
      {PRIORITY_LABELS[priority]}
    </PopoverTrigger>
    <PopoverContent className="w-32 p-1" align="end">
      {(["high", "medium", "low"] as const).map((p) => (
        <button
          key={p}
          className={cn(
            "w-full text-left text-sm px-2 py-1.5 rounded-sm hover:bg-accent",
            PRIORITY_STYLES[p]
          )}
          onClick={() => {
            setPriority(p);
            setIsPriorityOpen(false);
          }}
        >
          {PRIORITY_LABELS[p]}
        </button>
      ))}
    </PopoverContent>
  </Popover>

  {/* Date input (per D-09, UI-SPEC -- w-36 per spec) */}
  <input
    ref={dateInputRef}
    type="date"
    value={dueDate}
    onChange={(e) => setDueDate(e.target.value)}
    onKeyDown={(e) => {
      if (e.key === "Enter") {
        e.preventDefault();
        handleSubmit();
      } else if (e.key === "Escape") {
        e.preventDefault();
        resetForm();
      } else if (e.key === "Tab" && !e.shiftKey) {
        e.preventDefault();
        nameInputRef.current?.focus(); // Tab wraps back to name (D-12)
      } else if (e.key === "Tab" && e.shiftKey) {
        e.preventDefault();
        priorityButtonRef.current?.focus();
      }
    }}
    className="h-auto border-0 bg-transparent text-sm text-foreground w-36 shrink-0 focus:outline-none focus:ring-1 focus:ring-ring/50 rounded-sm px-1"
    aria-label="Due date"
  />
</div>
```

CRITICAL: `PopoverTrigger` uses `render` prop, NOT `asChild`. This project uses base-ui (Pitfall 1).

**Tab order wrapping (D-12, Pattern 6):**
- From name: Tab -> priority button (via `preventDefault()` + `priorityButtonRef.current?.focus()`)
- From priority: Tab -> date input (via `preventDefault()` + `dateInputRef.current?.focus()`)
- From date: Tab -> name input (via `preventDefault()` + `nameInputRef.current?.focus()`)
- Shift+Tab reverses: name -> date -> priority -> name

**Rendering logic:**
```tsx
return isExpanded ? <ExpandedRow /> : <CollapsedRow />;
```
