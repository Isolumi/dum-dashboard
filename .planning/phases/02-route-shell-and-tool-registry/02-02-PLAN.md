    **Step 1 — Create `src/routes/_layout.tsx`:**

    This is the pathless layout route. It wraps all child pages with the sidebar. The route ID is `/_layout` — it does NOT add a URL path segment.

    ```tsx
    import { Outlet, createFileRoute } from "@tanstack/react-router"
    import { SidebarProvider, SidebarInset } from "#/components/ui/sidebar"
    import { AppSidebar } from "#/components/AppSidebar"

    export const Route = createFileRoute("/_layout")({
      component: LayoutComponent,
    })

    function LayoutComponent() {
      return (
        <SidebarProvider>
          <AppSidebar />
          <SidebarInset>
            <Outlet />
          </SidebarInset>
        </SidebarProvider>
      )
    }
    ```

    Key details:
    - `createFileRoute("/_layout")` — the underscore prefix makes it pathless (no URL segment added)
    - `SidebarProvider` wraps everything — required by shadcn Sidebar to manage collapse state
    - `AppSidebar` renders the sidebar (from Plan 01)
    - `SidebarInset` wraps the main content area — the shadcn pattern for content beside a sidebar
    - `Outlet` renders the matched child route (overview page, future tool pages)
    - Import `Outlet` from `@tanstack/react-router` (NOT from `@tanstack/react-start` — per Research anti-pattern)

    **Step 2 — Create directory `src/routes/_layout/`:**

    mkdir -p src/routes/_layout

    **Step 3 — Delete `src/routes/index.tsx` and create `src/routes/_layout/index.tsx`:**

    The existing `src/routes/index.tsx` must be REMOVED (not just moved) because its `createFileRoute("/")` call is wrong for a layout child. The new file uses `createFileRoute("/_layout/")`.

    Create `src/routes/_layout/index.tsx` with the overview page:

    ```tsx
    import { createFileRoute } from "@tanstack/react-router"
    import { tools } from "#/tools/registry"

    export const Route = createFileRoute("/_layout/")({
      component: OverviewPage,
    })

    function OverviewPage() {
      return (
        <main className="p-6">
          <div className="grid grid-cols-1 gap-4 md:grid-cols-2 md:gap-6 lg:grid-cols-3">
            {tools.map((tool) => (
              <tool.BentoCard key={tool.id} tool={tool} />
            ))}
          </div>
        </main>
      )
    }
    ```

    Key details:
    - `createFileRoute("/_layout/")` — the trailing slash means this is the index route of the `_layout` directory, which resolves to URL `/`
    - Grid responsive breakpoints per UI-SPEC Bento Grid: 1 column default, 2 columns at md (768px), 3 columns at lg (1024px)
    - Grid gap: `gap-4` (16px) default, `md:gap-6` (24px) at md+ per UI-SPEC
    - Container padding: `p-6` (24px = lg token per UI-SPEC)
    - `tools.map` renders one `BentoCard` per registry entry — the overview page has NO hardcoded tool references (FOUN-04)
    - Each BentoCard receives `tool={tool}` prop matching `ComponentType<{ tool: ToolEntry }>`
    - No background colour class on `<main>` — the SidebarInset/page background inherits from `body` which uses `bg-background` (neutral-950) via `@layer base` in styles.css
