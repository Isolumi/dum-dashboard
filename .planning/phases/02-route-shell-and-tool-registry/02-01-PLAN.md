    Create `src/components/AppSidebar.tsx` with the following implementation:

    ```tsx
    import { Link, useMatchRoute } from "@tanstack/react-router"
    import { LayoutDashboard, LayoutGrid } from "lucide-react"
    import {
      Sidebar,
      SidebarContent,
      SidebarGroup,
      SidebarHeader,
      SidebarMenu,
      SidebarMenuButton,
      SidebarMenuItem,
      SidebarRail,
    } from "#/components/ui/sidebar"
    import { tools, type ToolEntry } from "#/tools/registry"

    function NavItem({ tool, isActive }: { tool: { label: string; route: string; icon: React.ComponentType }; isActive: boolean }) {
      const Icon = tool.icon
      return (
        <SidebarMenuButton
          render={<Link to={tool.route} />}
          isActive={isActive}
          tooltip={tool.label}
        >
          <Icon />
          <span>{tool.label}</span>
        </SidebarMenuButton>
      )
    }

    export function AppSidebar() {
      const matchRoute = useMatchRoute()

      // Overview is a hardcoded entry — not in the tools registry (Research Open Question 2)
      const isOverviewActive = Boolean(matchRoute({ to: "/" }))

      return (
        <Sidebar collapsible="icon">
          <SidebarHeader>
            <div className="flex items-center gap-2 px-4 py-3">
              <LayoutDashboard className="size-6 text-neutral-100" />
              <span className="text-sm font-semibold text-neutral-100">Dashboard</span>
            </div>
          </SidebarHeader>
          <SidebarContent>
            <SidebarGroup>
              <SidebarMenu>
                {/* Overview — hardcoded, always first */}
                <SidebarMenuItem>
                  <NavItem
                    tool={{ label: "Overview", route: "/", icon: LayoutGrid }}
                    isActive={isOverviewActive}
                  />
                </SidebarMenuItem>

                {/* Tool entries — driven by registry (FOUN-03) */}
                {tools.map((tool) => {
                  const isActive = Boolean(matchRoute({ to: tool.route }))
                  return (
                    <SidebarMenuItem key={tool.id}>
                      <NavItem tool={tool} isActive={isActive} />
                    </SidebarMenuItem>
                  )
                })}
              </SidebarMenu>
            </SidebarGroup>
          </SidebarContent>
          <SidebarRail />
        </Sidebar>
      )
    }
    ```

    **Key implementation details:**

    1. **Sidebar header:** Uses `LayoutDashboard` icon (24px via `size-6`) + "Dashboard" label (14px = `text-sm`, weight 600 = `font-semibold`) per UI-SPEC Sidebar Structure and Copywriting Contract. Padding `px-4 py-3` matches lg spacing token (sidebar header padding). Gap `gap-2` (8px = sm token per UI-SPEC icon-to-label gaps).

    2. **collapsible="icon":** Makes sidebar collapse to icon-only mode (56px width). SidebarRail provides the toggle control. Expanded width defaults to 240px (16rem) which is the shadcn default — matches UI-SPEC.

    3. **Overview nav item:** Hardcoded as the first sidebar entry (not in tools registry). Uses `LayoutGrid` icon. Per Research Open Question 2, this keeps the registry clean for actual tools. The Overview label and route are fixed strings per Copywriting Contract.

    4. **Registry-driven tool items:** Maps over `tools` from registry. Each gets a `NavItem` with `render={<Link to={tool.route} />}` for client-side navigation (NAV-02) and `isActive` from `useMatchRoute` (NAV-03). This satisfies FOUN-03 — the registry is the single source of truth.

    5. **render prop (NOT asChild):** This project uses base-ui (`"base": "base"` in components.json). Per `.claude/skills/shadcn/rules/base-vs-radix.md`, `SidebarMenuButton` uses `render` prop for custom element composition.

    6. **isActive + violet accent:** `isActive` prop on `SidebarMenuButton` triggers the shadcn active state styling. The `--sidebar-primary` CSS variable is already set to `oklch(0.65 0.2 270)` (violet-500) in `src/styles.css` from Phase 1. No additional CSS needed.

    7. **tooltip prop:** `tooltip={tool.label}` enables tooltips in collapsed (icon-only) state. shadcn's SidebarMenuButton handles this automatically.

    8. **No icon sizing classes:** Per shadcn SKILL.md icons rule, do not add `size-4` or `w-5 h-5` on icons inside `SidebarMenuButton` — the sidebar CSS handles icon sizing. Exception: the header icon uses `size-6` (className) because it is NOT inside a SidebarMenuButton — it is in SidebarHeader.

    9. **No hardcoded colour values:** All colours use Tailwind utility tokens (`text-neutral-100`). No oklch/hsl/rgb/hex in the file.
